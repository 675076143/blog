cmake_minimum_required(VERSION 3.25)
project(kwin_hdr_screencast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_AUTOMOC ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(QT_DEFAULT_MAJOR_VERSION 6)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)
find_package(Qt6 REQUIRED COMPONENTS Core Gui GuiPrivate Quick Widgets DBus)
find_package(KWin 6.8.1 EXACT REQUIRED)
find_package(KF6 REQUIRED COMPONENTS I18n CoreAddons)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PipeWire REQUIRED IMPORTED_TARGET libpipewire-0.3>=1.6)
find_package(Libdrm REQUIRED)
add_library(kwin ALIAS KWin::kwin)
add_compile_definitions(HAVE_PIPEWIRE_ST2084=1)
include_directories(/usr/include/kwin ${KWIN_SOURCE_DIR}/src)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
add_subdirectory(${KWIN_SOURCE_DIR}/src/plugins/screencast screencast)

set_target_properties(screencast PROPERTIES OUTPUT_NAME "screencast-hdr-dynamic")
